  (interactive)

         (file (buffer-file-name)))

(defun my/unpop-to-mark ()
  "Unpop off the mark ring (move forward to a newer position). Does nothing
if the mark ring is empty."
  (interactive)
  (deactivate-mark)
  (when mark-ring
    (setq mark-ring (cons (copy-marker (mark-marker)) mark-ring))
    (set-marker (mark-marker) (car (last mark-ring)) (current-buffer))
    (setq mark-ring (nbutlast mark-ring))
    (goto-char (marker-position (car (last mark-ring))))))

  (pop-to-mark-command))

** Symbol/word occurrence collector (Meow-extracted, ~my/jump-to-symbol~)
Extracted from ~meow--prompt-symbol-and-words~ (meow-util.el:284–305, meow
1.5.0, GPLv3). Scans a region, collects every symbol/word as a completion
candidate (each presented as its word-boundary regexp ~\\_<...\\_>~ when
~my/symbol-collect-sanitize~ is non-nil), and lets the user pick one via
~completing-read~. The helper returns the chosen symbol or its regexp; bound
consumer ~my/jump-to-symbol~ jumps to the chosen symbol's next occurrence.
Deps: builtins only (~save-mark-and-excursion~, ~re-search-forward~,
~match-string-no-properties~, ~regexp-quote~, ~delete-dups~, ~completing-read~,
~string-empty-p~). Layout-independent, Dvorak-safe. Bound to ~f~ (find word /
symbol) in normal + motion states.
#+begin_src emacs-lisp
(defcustom my/symbol-collect-min-length 1
  "Minimal length for a symbol/word to be collected as a completion candidate."
  :type 'integer
  :group 'manifolding-keyboard)

(defcustom my/symbol-collect-sanitize t
  "If non-nil, present each candidate as its ~\\_<...\\_>~ word-boundary regexp."
  :type 'boolean
  :group 'manifolding-keyboard)

(defun my/prompt-symbol-and-words (prompt beg end &optional disallow-empty)
  "Completion with PROMPT for symbols and words from BEG to END.

Returns the chosen symbol, or — when ~my/symbol-collect-sanitize~ is non-nil —
its word-boundary regexp ~\\_<SYMBOL\\_>~. If DISALLOW-EMPTY, requests again
until a non-empty selection is made."
  (let ((completions))
    (save-mark-and-excursion
      (goto-char beg)
      (while (re-search-forward "\\_<\\(\\sw\\|\\s_\\)+\\_>" end t)
        (let ((result (match-string-no-properties 0)))
          (when (>= (length result) my/symbol-collect-min-length)
            (if my/symbol-collect-sanitize
                (push (cons result (format "\\_<%s\\_>" (regexp-quote result)))
                      completions)
              (push (format "\\_<%s\\_>" (regexp-quote result))
                    completions))))))
    (setq completions (delete-dups completions))
    (let ((selected (completing-read prompt completions nil nil)))
      (while (and (string-empty-p selected) disallow-empty)
        (setq selected (completing-read
                        (concat "[Input must be non-empty] " prompt)
                        completions nil nil)))
      (if my/symbol-collect-sanitize
          (or (cdr (assoc selected completions))
              (regexp-quote selected))
        selected))))

(defun my/jump-to-symbol ()
  "Pick any symbol/word in the buffer, then jump to its next occurrence.

Collects symbols/words from the whole buffer via ~completing-read~ and moves
point to the next match (wrapping to the top on reaching the buffer end)."
  (interactive)
  (let ((target (my/prompt-symbol-and-words "Jump to symbol: "
                                             (point-min) (point-max))))
    (when target
      (unless (re-search-forward target nil t)
        (goto-char (point-min))
        (unless (re-search-forward target nil t)
          (message "No occurrence of %s" target))))))
#+end_src

** Balanced-delimiter bounds finder (Meow-extracted, ~my/select-inside-delimiters~)
Extracted from ~meow--thing-pair-function~ / ~meow--thing-parse-pair-search~
(meow-thing.el:134–189, meow 1.5.0, GPLv3). Finds the ~(beg . end)~ of a
balanced delimiter pair by depth-counting push/pop tokens in both directions.
The ported helpers are general (any push/pop regexp tokens); the bound consumer
~my/select-inside-delimiters~ finds the innermost enclosing ~() [] {} <>~ pair
around point and activates its inner CONTENT as the region (visual state).
Deps: builtins + cl-lib (~save-mark-and-excursion~, ~re-search-forward~,
~re-search-backward~, ~match-beginning~, ~match-end~, ~regexp-quote~,
~cl-incf~, ~cl-decf~). Layout-independent, Dvorak-safe. Bound to ~b~ (block /
bracket) in normal + motion states.
#+begin_src emacs-lisp
(defun my/thing-parse-pair-search (push-token pop-token back near)
  "Search for the nearest PUSH-TOKEN or POP-TOKEN (regexps) around point.

When BACK is non-nil search backward, else forward. Returns (push|pop . POS)
with POS at the delimiter boundary; moves point to the token start for the
next iteration. Returns nil if neither token is found."
  (let* ((search-fn (if back #'re-search-backward #'re-search-forward))
         (match-fn (if back #'match-end #'match-beginning))
         (cmp-fn (if back #'> #'<))
         (push-next-pos nil)
         (pop-next-pos nil)
         (push-pos (save-mark-and-excursion
                     (when (funcall search-fn push-token nil t)
                       (setq push-next-pos (point))
                       (if near (funcall match-fn 0) (point)))))
         (pop-pos (save-mark-and-excursion
                    (when (funcall search-fn pop-token nil t)
                      (setq pop-next-pos (point))
                      (if near (funcall match-fn 0) (point))))))
    (cond
     ((and (not pop-pos) (not push-pos)) nil)
     ((not pop-pos) (goto-char push-next-pos) (cons 'push push-pos))
     ((not push-pos) (goto-char pop-next-pos) (cons 'pop pop-pos))
     ((funcall cmp-fn push-pos pop-pos)
      (goto-char push-next-pos) (cons 'push push-pos))
     (t
      (goto-char pop-next-pos) (cons 'pop pop-pos)))))

(defun my/thing-pair-function (push-token pop-token near)
  "Return (beg . end) of the balanced pair of regexps PUSH-TOKEN/POP-TOKEN
surrounding point, or nil. With NEAR, boundaries hug the delimiters; else beg
is the opening token's position and end is just past the closing token."
  (let* ((found nil)
         (depth 0)
         (beg (save-mark-and-excursion
                (prog1
                    (let ((case-fold-search nil))
                      (while (and (<= depth 0)
                                  (setq found (my/thing-parse-pair-search
                                               push-token pop-token t near)))
                        (let ((push-or-pop (car found)))
                          (if (eq 'push push-or-pop)
                              (cl-incf depth)
                            (cl-decf depth))))
                      (when (> depth 0) (cdr found)))
                  (setq depth 0 found nil))))
         (end (save-mark-and-excursion
                (let ((case-fold-search nil))
                  (while (and (>= depth 0)
                              (setq found (my/thing-parse-pair-search
                                           push-token pop-token nil near)))
                    (let ((push-or-pop (car found)))
                      (if (eq 'push push-or-pop)
                          (cl-incf depth)
                        (cl-decf depth))))
                  (when (< depth 0) (cdr found))))))
    (when (and beg end)
      (cons beg end))))

(defun my/select-inside-delimiters ()
  "Activate the INNER content of the innermost enclosing ~() [] {} <>~ around point.

Finds the nearest balanced delimiter pair surrounding point, sets point and
mark to the text strictly between the delimiters, and enters visual state."
  (interactive)
  (let ((pairs '(("\\\\(" . "\\\\)") ("\\\\[" . "\\\\]") ("{" . "}") ("<" . ">")))
        (best nil))
    (dolist (p pairs)
      (let* ((found (my/thing-pair-function (car p) (cdr p) nil))
             (beg (and found (car found)))
             (end (and found (cdr found))))
        (when (and beg end (<= beg (point) end)
                   (or (null best) (> beg (car best))))
          (setq best (cons beg end)))))
    (if best
        (progn
          (goto-char (car best))
          (push-mark (1+ (point)) t t)
          (goto-char (1- (cdr best)))
          (activate-mark)
          (manifolding-keyboard-set-state "visual"))
      (message "No enclosing delimiter pair"))))
#+end_src

** Syntax-context predicates (Meow-extracted, ~my/select-syntax-context~)
Extracted from ~meow--in-string-p~ / ~meow--in-comment-p~ (meow-util.el:251–261,

#+begin_src emacs-lisp

(defun my/in-comment-p (&optional pos)
  "Return whether POS (default point) is inside a comment."
  (save-mark-and-excursion
    (when pos (goto-char pos))
    (nth 4 (syntax-ppss))))

(defun my/select-syntax-context ()
  "Select the enclosing STRING or COMMENT containing point (visual state).

If point is inside a string literal, selects the whole literal; if inside a
comment, selects the whole comment (block comments to their terminator, line
comments to end of line). Otherwise no-op with a message."
  (interactive)
  (let ((ppss (syntax-ppss)))
    (cond
     ((my/in-string-p)
      (let ((start (nth 8 ppss)))
        (goto-char start)
        (forward-sexp 1)
        (push-mark (1+ start) t t)
        (activate-mark)
        (manifolding-keyboard-set-state "visual")))
     ((my/in-comment-p)
      (let* ((start (nth 8 ppss))
             (end (save-mark-and-excursion
                    (goto-char start)
                    (condition-case nil
                        (progn
                          (skip-syntax-backward " ")
                          (if comment-end-skip
                              (progn
                                (re-search-forward comment-end-skip nil t)
                                (if (and comment-end (not (equal comment-end "")))
                                    (progn (goto-char match-beginning) (point))
                                  (point)))
                            (line-end-position)))
                      (error (line-end-position))))))
        (goto-char start)
        (push-mark start t t)
        (goto-char end)
        (activate-mark)
        (manifolding-keyboard-set-state "visual")))
     (t
      (message "Point is not inside a string or comment")))))

(defun my/ensure-visible ()
  "Expose invisible overlays at point so folded/hidden content becomes visible.

Iterates the overlays touching the char before point and, for each that is
invisible AND carries an ~isearch-open-invisible~ handler, invokes that handler
(the org/outline/isearch mechanism that actually unfolds)."
  (let ((overlays (overlays-at (1- (point))))
        ov expose)
    (while (setq ov (pop overlays))
      (if (and (invisible-p (overlay-get ov 'invisible))
               (setq expose (overlay-get ov 'isearch-open-invisible)))
          (funcall expose ov)))))

  "Reveal folded or hidden content at point."
  (interactive)
  (my/ensure-visible))

(defun my/get-indent ()
  "Get the indentation column of the current line."
  (save-mark-and-excursion
    (back-to-indentation)
    (- (point) (line-beginning-position))))

(defun my/empty-line-p ()
  "Return whether the current line is blank (only whitespace)."
  (string-match-p "^ *$" (buffer-substring-no-properties
                          (line-beginning-position)
                          (line-end-position))))

(defun my/indent-to-match ()
  "Indent the current line to match the nearest preceding non-blank line.

Scans upward for the previous non-empty line and sets this line's leading
whitespace to that line's indentation column."
  (interactive)
  (let ((target nil))
    (save-excursion
      (goto-char (line-beginning-position))
      (while (and (not target) (not (bobp)))
        (forward-line -1)
        (unless (my/empty-line-p)
          (setq target (my/get-indent)))))
    (if (not target)
        (message "No preceding non-blank line to match")
      (let ((line-start (line-beginning-position)))
        (goto-char line-start)
        (delete-region line-start (progn (skip-chars-forward " \t") (point)))
        (indent-to target)
        (back-to-indentation)))))

** Smart line motion (Boon-extracted, ~my/line-beginning~ / ~my/line-end~)
Extracted from ~boon-beginning-of-line~ (boon-moves.el:183–193) and
~boon-end-of-line~ (boon-moves.el:196–216) plus their helpers
~boon-visible-beginning-of-line~ (boon-moves.el:174–180) and
~boon-looking-at-line-comment-start-p~ (boon-utils.el:55–60), all from Boon
(GPLv3). Smart home/end cycling: BOL goes to the first non-whitespace char,
else the true beginning of line (skipping folded outline text); EOL cycles
last-code-char → last-non-blank-char → true end-of-line, honoring comment
starts. Reuses the previously imported ~my/in-string-p~ (Meow #06) instead of
Boon's ~boon-in-string-p~. Deps: builtins only. Layout-independent,
Dvorak-safe. Bound to ~y~ (line-beginning) and ~z~ (line-end) in normal +
motion states.
#+begin_src emacs-lisp

(defun my/line-beginning ()
  "Move point to first non-whitespace on the line; else to true line start.
Skips folded (invisible) outline text at the beginning of line."
  (interactive)
  (let ((oldpos (point)))
    (back-to-indentation)
    (when (or (and (fboundp 'outline-invisible-p)
                   (outline-invisible-p))
              (= oldpos (point)))
      (beginning-of-line)
      (while (bound-and-true-p outline-invisible-p)
        (backward-char 1)
        (beginning-of-line 1)))))

(defun my/line-end ()
  "Move point to the end of the line, cycling through smart positions.
Cycles: last non-comment char, then last non-blank char, then true end of line."
  (interactive)
  (let* ((orig (point))
         (orig-eol (eolp))
         (progress (lambda () (and (not (bolp)) (or orig-eol (> (point) orig))))))
    (beginning-of-line)
    (while (not (or (my/line-comment-start-p) (eolp)))
      (forward-char))
    (skip-chars-backward "\n\t " (line-beginning-position))
    (unless (funcall progress)
      (end-of-line)
      (skip-chars-backward "\n\t " (line-beginning-position))
      (unless (funcall progress)
        (end-of-line)))))
#+end_src

  "Act on the thing at point.

type-appropriate default action (Embark-extracted). The per-type actions REUSE
existing M-K functions where they already cover a type — no parallel duplicates:

(defun my/act-all-markers (thing)
  "Return markers for every occurrence of THING in the buffer, in order."
  (let (markers)
    (save-excursion
      (goto-char (point-min))
      (while (re-search-forward (regexp-quote thing) nil t)
        (push (cons (copy-marker (match-beginning 0))
                    (copy-marker (match-end 0)))
              markers)))
    (nreverse markers)))

(defun my/act-all-delete (thing markers)
  "Delete every occurrence of THING marked by MARKERS."
  (dolist (m markers)
    (let ((beg (car m)) (end (cdr m)))
      (when (and (marker-buffer beg) (<= (point-min) beg end (point-max)))
        (delete-region beg end))))
  (message "Deleted all %s" thing))

(defun my/act-all-copy (thing markers)
  "Copy every occurrence of THING marked by MARKERS to the kill ring."
  (let ((text (mapconcat (lambda (m)
                           (buffer-substring-no-properties (car m) (cdr m)))
                         markers "\n")))
    (kill-new text)
    (message "Copied %s: %d occurrence(s)" thing (length markers))))
