#+title: Manifolding Keyboard — Visual State
#+priority: 20

Keys specific to the visual state. Shared bindings live in universal.org.

** Owned functions
#+begin_src emacs-lisp
(defun my/visual-normal-state ()
  "Deactivate mark and switch to normal state."
  (interactive)
  (deactivate-mark)
  (manifolding-keyboard-set-state "normal"))
#+end_src
** Bindings
#+begin_src emacs-lisp
(manifolding-keyboard-define-keys
  :states '("visual")
  :bind
  `(
    ("ESC" . my/visual-normal-state)
    ("a" . mark-word)
    ("c" . consult-yank-pop)
    ("d" . kill-region)
    ("e" . mark-line)
    ("h" . beginning-of-line)
    ("i" . mark-inner-region)
    ("n" . kill-ring-save)
    ("o" . mark-sexp)
    ("t" . end-of-line)
    ("u" . rectangle-mark-mode)
))
#+end_src
