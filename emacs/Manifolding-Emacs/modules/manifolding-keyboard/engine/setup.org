** Init State
#+begin_src emacs-lisp
(setq manifolding-keyboard-init-state-fn
      (lambda ()
        (cond
         ((minibufferp) nil)
         ((derived-mode-p 'org-mode)    "org")
         ((derived-mode-p 'dired-mode)  "dired")
         ((derived-mode-p 'vterm-mode)  "vterm")
         ((derived-mode-p 'ibuffer-mode) "ibuffer")
         ((derived-mode-p 'magit-mode)  nil)
         ((derived-mode-p 'vundo-mode)  nil)
         ((buffer-local-value 'buffer-read-only (current-buffer)) "motion")
         (t "normal"))))
(manifolding-keyboard-setup)
#+end_src
